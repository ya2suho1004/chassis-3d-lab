```ts
// src/app/entry.ts
// 입구 판단(설계 2026-11-09-qr-entry). 코드는 주소의 # 뒤에만 둔다. # 뒤는 서버 요청에 포함되지 않아 접속 로그에 남지 않는다.
import { validateStudentCode } from './session'

export type Entry =
  | { kind: 'card'; code: string; classLetter: string }
  | { kind: 'noRecordCard'; classLetter: string }
  | { kind: 'teacher' }
  | { kind: 'demo' }
  | { kind: 'plain' }

const LETTER_RE = /^[A-Z]$/

export const classLetterOf = (code: string) => code.charAt(0)

export function parseEntry(hash: string): Entry {
  const h = hash.replace(/^#/, '')
  if (h === 'teacher') return { kind: 'teacher' }
  if (h === 'demo') return { kind: 'demo' }
  const p = new URLSearchParams(h)
  const c = p.get('c')
  if (c !== null) {
    const v = validateStudentCode(c)
    return v.ok ? { kind: 'card', code: v.code, classLetter: classLetterOf(v.code) } : { kind: 'plain' }
  }
  const n = p.get('n')?.trim().toUpperCase()
  if (n && LETTER_RE.test(n)) return { kind: 'noRecordCard', classLetter: n }
  return { kind: 'plain' }
}

export const cardUrl = (base: string, code: string) => `${base}#c=${code}`
export const noRecordCardUrl = (base: string, letter: string) => `${base}#n=${letter}`

/** 코드가 주소창과 방문 기록에 남지 않게 지운다 */
export function clearCardFromAddress(loc: Pick<Location, 'pathname' | 'search'>, hist: Pick<History, 'replaceState'>) {
  hist.replaceState(null, '', loc.pathname + loc.search)
}
```

```ts
// src/teacher/qr.ts
// 입장 카드용 QR 생성(설계 2026-11-09-qr-entry). 새 패키지를 쓰지 않고, 코드를 외부 서비스로 보내지 않는다.
// 범위: 버전 1~6, 오류 정정 M, 바이트 방식. 주소 길이는 106바이트까지.
const VERSIONS: Record<number, { total: number; ecPerBlock: number; blocks: number; align: number | null }> = {
  1: { total: 26, ecPerBlock: 10, blocks: 1, align: null },
  2: { total: 44, ecPerBlock: 16, blocks: 1, align: 18 },
  3: { total: 70, ecPerBlock: 26, blocks: 1, align: 22 },
  4: { total: 100, ecPerBlock: 18, blocks: 2, align: 26 },
  5: { total: 134, ecPerBlock: 24, blocks: 2, align: 30 },
  6: { total: 172, ecPerBlock: 16, blocks: 4, align: 34 },
}

function gfMul(x: number, y: number): number {
  let z = 0
  for (let i = 7; i >= 0; i--) {
    z = (z << 1) ^ ((z >>> 7) * 0x11d)
    z ^= ((y >>> i) & 1) * x
  }
  return z & 0xff
}

function rsDivisor(degree: number): number[] {
  const result = new Array<number>(degree).fill(0)
  result[degree - 1] = 1
  let root = 1
  for (let i = 0; i < degree; i++) {
    for (let j = 0; j < degree; j++) {
      result[j] = gfMul(result[j]!, root)
      if (j + 1 < degree) result[j] = result[j]! ^ result[j + 1]!
    }
    root = gfMul(root, 2)
  }
  return result
}

export function rsRemainder(data: number[], degree: number): number[] {
  const divisor = rsDivisor(degree)
  const result = new Array<number>(degree).fill(0)
  for (const b of data) {
    const factor = b ^ result.shift()!
    result.push(0)
    divisor.forEach((d, i) => { result[i] = result[i]! ^ gfMul(d, factor) })
  }
  return result
}

/** 오류 정정 M(00) + 마스크의 15비트 형식 정보 */
export function formatBits(mask: number): number {
  const data = mask // M의 지시 비트가 00이라 마스크 값과 같다
  let rem = data
  for (let i = 0; i < 10; i++) rem = (rem << 1) ^ ((rem >>> 9) * 0x537)
  return ((data << 10) | rem) ^ 0x5412
}

function codewords(bytes: Uint8Array, version: number): number[] {
  const v = VERSIONS[version]!
  const capacity = v.total - v.ecPerBlock * v.blocks
  const bits: number[] = []
  const put = (val: number, len: number) => { for (let i = len - 1; i >= 0; i--) bits.push((val >>> i) & 1) }
  put(4, 4)
  put(bytes.length, 8)
  bytes.forEach((b) => put(b, 8))
  put(0, Math.min(4, capacity * 8 - bits.length))
  put(0, (8 - (bits.length % 8)) % 8)
  const data: number[] = []
  for (let i = 0; i < bits.length; i += 8) data.push(bits.slice(i, i + 8).reduce((a, b) => (a << 1) | b, 0))
  for (let pad = 0xec; data.length < capacity; pad ^= 0xec ^ 0x11) data.push(pad)
  // 이 범위의 M 등급은 블록 길이가 모두 같다
  const dataPerBlock = capacity / v.blocks
  const dataBlocks: number[][] = []
  const ecBlocks: number[][] = []
  for (let b = 0; b < v.blocks; b++) {
    const block = data.slice(b * dataPerBlock, (b + 1) * dataPerBlock)
    dataBlocks.push(block)
    ecBlocks.push(rsRemainder(block, v.ecPerBlock))
  }
  const out: number[] = []
  for (let i = 0; i < dataPerBlock; i++) dataBlocks.forEach((bl) => out.push(bl[i]!))
  for (let i = 0; i < v.ecPerBlock; i++) ecBlocks.forEach((bl) => out.push(bl[i]!))
  return out
}

const MASKS: Array<(x: number, y: number) => boolean> = [
  (x, y) => (x + y) % 2 === 0,
  (_, y) => y % 2 === 0,
  (x) => x % 3 === 0,
  (x, y) => (x + y) % 3 === 0,
  (x, y) => (Math.floor(x / 3) + Math.floor(y / 2)) % 2 === 0,
  (x, y) => ((x * y) % 2) + ((x * y) % 3) === 0,
  (x, y) => (((x * y) % 2) + ((x * y) % 3)) % 2 === 0,
  (x, y) => (((x + y) % 2) + ((x * y) % 3)) % 2 === 0,
]

function penalty(m: boolean[][]): number {
  const n = m.length
  let score = 0
  const lineScore = (line: boolean[]) => {
    let s = 0, run = 1
    for (let i = 1; i <= n; i++) {
      if (i < n && line[i] === line[i - 1]) run++
      else { if (run >= 5) s += run - 2; run = 1 }
    }
    const text = line.map((v) => (v ? '1' : '0')).join('')
    for (const pat of ['10111010000', '00001011101']) for (let at = text.indexOf(pat); at >= 0; at = text.indexOf(pat, at + 1)) s += 40
    return s
  }
  let dark = 0
  for (let y = 0; y < n; y++) {
    score += lineScore(m[y]!)
    score += lineScore(m.map((row) => row[y]!))
    for (let x = 0; x < n; x++) {
      if (m[y]![x]) dark++
      if (x + 1 < n && y + 1 < n && m[y]![x] === m[y]![x + 1] && m[y]![x] === m[y + 1]![x] && m[y]![x] === m[y + 1]![x + 1]) score += 3
    }
  }
  score += Math.floor(Math.abs((dark * 100) / (n * n) - 50) / 5) * 10
  return score
}

export function qrMatrix(text: string): boolean[][] {
  const bytes = new TextEncoder().encode(text)
  const version = Object.keys(VERSIONS).map(Number).find((ver) => {
    const v = VERSIONS[ver]!
    return bytes.length + 2 <= v.total - v.ecPerBlock * v.blocks
  })
  if (!version) throw new Error('QR_TOO_LONG')
  const n = version * 4 + 17
  const mod = Array.from({ length: n }, () => new Array<boolean>(n).fill(false))
  const fixed = Array.from({ length: n }, () => new Array<boolean>(n).fill(false))
  const set = (x: number, y: number, on: boolean) => { mod[y]![x] = on; fixed[y]![x] = true }

  for (let i = 0; i < n; i++) { set(6, i, i % 2 === 0); set(i, 6, i % 2 === 0) }
  const finder = (cx: number, cy: number) => {
    for (let dy = -4; dy <= 4; dy++) for (let dx = -4; dx <= 4; dx++) {
      const x = cx + dx, y = cy + dy
      if (x < 0 || y < 0 || x >= n || y >= n) continue
      const d = Math.max(Math.abs(dx), Math.abs(dy))
      set(x, y, d !== 2 && d !== 4)
    }
  }
  finder(3, 3); finder(n - 4, 3); finder(3, n - 4)
  const align = VERSIONS[version]!.align
  if (align !== null) {
    for (let dy = -2; dy <= 2; dy++) for (let dx = -2; dx <= 2; dx++) set(align + dx, align + dy, Math.max(Math.abs(dx), Math.abs(dy)) !== 1)
  }
  const drawFormat = (bits: number) => {
    const bit = (i: number) => ((bits >>> i) & 1) === 1
    for (let i = 0; i <= 5; i++) set(8, i, bit(i))
    set(8, 7, bit(6)); set(8, 8, bit(7)); set(7, 8, bit(8))
    for (let i = 9; i < 15; i++) set(14 - i, 8, bit(i))
    for (let i = 0; i < 8; i++) set(n - 1 - i, 8, bit(i))
    for (let i = 8; i < 15; i++) set(8, n - 15 + i, bit(i))
    set(8, n - 8, true)
  }
  drawFormat(0) // 자리 확보

  const data = codewords(bytes, version)
  let i = 0
  for (let right = n - 1; right >= 1; right -= 2) {
    if (right === 6) right = 5
    for (let vert = 0; vert < n; vert++) {
      for (let j = 0; j < 2; j++) {
        const x = right - j
        const up = ((right + 1) & 2) === 0
        const y = up ? n - 1 - vert : vert
        if (!fixed[y]![x] && i < data.length * 8) {
          mod[y]![x] = ((data[i >>> 3]! >>> (7 - (i & 7))) & 1) === 1
          i++
        }
      }
    }
  }

  const applyMask = (mask: number) => {
    for (let y = 0; y < n; y++) for (let x = 0; x < n; x++) if (!fixed[y]![x] && MASKS[mask]!(x, y)) mod[y]![x] = !mod[y]![x]
  }
  let best = 0, bestScore = Infinity
  for (let mask = 0; mask < 8; mask++) {
    applyMask(mask); drawFormat(formatBits(mask))
    const s = penalty(mod)
    if (s < bestScore) { best = mask; bestScore = s }
    applyMask(mask)
  }
  applyMask(best); drawFormat(formatBits(best))
  return mod
}
```

```ts
import type { UsagePhase } from '../domain/record'
import type { DifficultyMode } from '../domain/taskEngine'
import type { ApprovedTask } from '../domain/types'
import type { StoredRecord } from '../records/repository'
import type { ServerConfig } from '../teacher/serverAdmin'
import { lessonAttemptRole, lessonMode } from './lesson'
import { createDemoSession } from './session'
import { applySessionRules, sessionDef } from './sessions'
import { caseKey } from './teacherPreset'

export interface ClassTodayRow {
  class_letter: string
  session_code: string
  case_ids: string[]
  usage_phase: UsagePhase
  difficulty_mode: 'guided' | 'unguided'
  vehicle_entry_direction: 'north' | 'south' | null
}
export type TodayFailure = 'NO_ROW' | 'UNKNOWN_SESSION' | 'PENDING_SESSION' | 'BAD_TASK_COUNT' | 'UNKNOWN_TASK' | 'NO_SNAPSHOT'
export interface TodayClass { sessionCode: string; usagePhase: UsagePhase; mode: DifficultyMode; direction: 'north' | 'south' | null; tasks: ApprovedTask[] }

export function sessionForTask(today: TodayClass, task: ApprovedTask, code: string) {
  return createDemoSession({ studentCode: code, sessionCode: today.sessionCode, attemptRole: lessonAttemptRole(today.usagePhase), usagePhase: today.usagePhase, vehicleEntryDirection: today.direction, task, difficultyMode: today.mode })
}

export function judgeToday(row: ClassTodayRow | null, tasks: ApprovedTask[]): { ok: true; today: TodayClass } | { ok: false; reason: TodayFailure } {
  const fail = (reason: TodayFailure) => ({ ok: false as const, reason })
  if (!row) return fail('NO_ROW')
  const def = sessionDef(row.session_code)
  if (!def) return fail('UNKNOWN_SESSION')
  if (def.pending) return fail('PENDING_SESSION')
  const ids = row.case_ids
  if (!Array.isArray(ids) || ids.length < 1 || ids.length > 2 || new Set(ids).size !== ids.length) return fail('BAD_TASK_COUNT')
  const picked = ids.map((id) => tasks.find((t) => caseKey(t) === id))
  if (picked.some((t) => !t)) return fail('UNKNOWN_TASK')
  const fixed = applySessionRules(row.session_code, { usagePhase: row.usage_phase, difficultyMode: row.difficulty_mode as DifficultyMode })
  const today: TodayClass = { sessionCode: row.session_code, usagePhase: fixed.usagePhase, mode: lessonMode(fixed.usagePhase, fixed.difficultyMode), direction: row.vehicle_entry_direction ?? null, tasks: picked as ApprovedTask[] }
  try {
    for (const t of today.tasks) sessionForTask(today, t, 'D-01')
  } catch {
    return fail('NO_SNAPSHOT')
  }
  return { ok: true, today }
}

/** 같은 코드·세션의 확정 기록에서 끝낸 과제를 찾는다. 기록은 과제 정의 판으로 과제를 가리킨다 */
export function doneCaseIds(records: StoredRecord[], code: string, sessionCode: string, tasks: ApprovedTask[]): string[] {
  const out: string[] = []
  for (const r of records) {
    if (r.payload.anonymousStudentCode !== code || r.payload.sessionCode !== sessionCode) continue
    const t = tasks.find((x) => x.taskDefinitionVersion === r.payload.versions.taskDefinitionVersion)
    if (t && !out.includes(caseKey(t))) out.push(caseKey(t))
  }
  return out
}

export const remainingTasks = (today: TodayClass, done: string[]) => today.tasks.filter((t) => !done.includes(caseKey(t)))

/** 입장할 때 고를 과제. 모두 끝냈으면 다시 할 수 있게 전체를 돌려준다 */
export function nextTasks(today: TodayClass, done: string[]) {
  const left = remainingTasks(today, done)
  return left.length > 0 ? left : today.tasks
}

const COLUMNS = 'class_letter,session_code,case_ids,usage_phase,difficulty_mode,vehicle_entry_direction'

async function readRows(cfg: ServerConfig, query: string, f: typeof fetch, timeoutMs: number): Promise<ClassTodayRow[]> {
  const ctl = new AbortController()
  const timer = setTimeout(() => ctl.abort(), timeoutMs)
  try {
    const r = await f(`${cfg.url}/rest/v1/class_today?${query}`, { headers: { apikey: cfg.anonKey, Authorization: `Bearer ${cfg.anonKey}` }, signal: ctl.signal })
    if (!r.ok) throw new Error(`CLASS_TODAY_${r.status}`)
    return (await r.json()) as ClassTodayRow[]
  } finally {
    clearTimeout(timer)
  }
}

export async function fetchClassToday(cfg: ServerConfig, letter: string, f: typeof fetch = fetch, timeoutMs = 5000): Promise<ClassTodayRow | null> {
  const rows = await readRows(cfg, `class_letter=eq.${encodeURIComponent(letter)}&select=${COLUMNS}`, f, timeoutMs)
  return rows[0] ?? null
}

export const listClassToday = (cfg: ServerConfig, f: typeof fetch = fetch, timeoutMs = 5000) =>
  readRows(cfg, `select=${COLUMNS}&order=class_letter.asc`, f, timeoutMs)
```

```ts
  BAD_TASK_COUNT: '오늘 과제 수가 맞지 않습니다',
  UNKNOWN_TASK: '오늘 과제를 찾지 못했습니다',
  NO_SNAPSHOT: '이 세션과 과제의 조합을 쓸 수 없습니다',
}

export async function decideEntry(i: {
  /** null이면 불참 카드: 기록·초안을 보지 않는다 */
  code: string | null
  classLetter: string
  tasks: ApprovedTask[]
  fetchRow: ((letter: string) => Promise<ClassTodayRow | null>) | null
  repo: Pick<Repository, 'listDrafts' | 'listRecords' | 'getCurrentRecordId' | 'clearCurrent' | 'loadDraft'> | null
}): Promise<EntryDecision> {
  if (!i.fetchRow) return { kind: 'fail', reason: 'NO_SERVER' }
  let row: ClassTodayRow | null
  try {
    row = await i.fetchRow(i.classLetter)
  } catch {
    return { kind: 'fail', reason: 'NETWORK' }
  }
  const judged = judgeToday(row, i.tasks)
  if (!judged.ok) return { kind: 'fail', reason: judged.reason }
  const today = judged.today
  if (i.code === null || !i.repo) return { kind: 'go', today, done: [] }

  const cur = i.repo.getCurrentRecordId()
  if (cur) {
    const curDraft = await i.repo.loadDraft(cur)
    if (curDraft?.session.anonymousStudentCode !== i.code) i.repo.clearCurrent()
  }
  const done = doneCaseIds(await i.repo.listRecords(), i.code, today.sessionCode, i.tasks)
  const found = findDraftForCode(await i.repo.listDrafts(), i.code, today.sessionCode)
  return found ? { kind: 'resume', draft: found, today, done } : { kind: 'go', today, done }
}
```

```ts
  if (!s) return 'AI 상태 모름'

export function todaySummary(o: {

  classes: ClassTodayRow[] | undefined
  tasks: ApprovedTask[]
  sessions: SessionRow[] | null
  counts: { records: number; outbox: number } | null

  const warnings: string[] = []
```

```tsx
import { useMemo, useState } from 'react'
import { cardUrl, noRecordCardUrl } from '../app/entry'
import { qrMatrix, qrPath } from '../teacher/qr'

export interface CardItem { label: string; url: string }
const two = (n: number) => String(n).padStart(2, '0')

export function buildCards(base: string, letter: string, from: number, to: number, noRecord: number): CardItem[] {
  if (!/^[A-Z]$/.test(letter)) throw new Error('학급 글자는 영문 대문자 하나입니다.')
  if (!Number.isInteger(from) || !Number.isInteger(to) || from < 1 || to < from || to > 89) throw new Error('참여 번호는 1~89 사이로 적어 주세요. 90번대는 불참 카드용입니다.')
  if (!Number.isInteger(noRecord) || noRecord < 0 || noRecord > 10) throw new Error('불참 카드는 0~10장으로 적어 주세요.')
  const cards: CardItem[] = []
  for (let n = from; n <= to; n++) cards.push({ label: `${letter}-${two(n)}`, url: cardUrl(base, `${letter}-${two(n)}`) })
  for (let i = 0; i < noRecord; i++) cards.push({ label: `${letter}-${90 + i}`, url: noRecordCardUrl(base, letter) })
  return cards
}

function Card({ card }: { card: CardItem }) {
```
